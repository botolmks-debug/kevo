function momenHariIni() {
  const kini = new Date();
  const penuh = kini.toISOString().slice(0, 10);
  const pendek = penuh.slice(5);

  for (let i = 0; i < MOMEN_KHUSUS.length; i++) {
    const m = MOMEN_KHUSUS[i];
    if (penuh >= m.mulai && penuh <= m.selesai) return m.teks;
  }
  for (let i = 0; i < MOMEN.length; i++) {
    const m = MOMEN[i];
    if (pendek >= m.mulai && pendek <= m.selesai) return m.teks;
  }
  return '';
}

function buatSlug(teks) {
  const dasar = String(teks)
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '')
    .slice(0, 60);
  const acak = Math.random().toString(36).slice(2, 6);
  return dasar + '-' + acak;
}

function ambilJson(teks) {
  let bersih = String(teks).trim();
  bersih = bersih.replace(/```json/gi, '').replace(/```/g, '').trim();
  const awal = bersih.indexOf('{');
  const akhir = bersih.lastIndexOf('}');
  if (awal !== -1 && akhir !== -1) bersih = bersih.slice(awal, akhir + 1);
  return JSON.parse(bersih);
}

function sudahDiposting(k) {
  if (!k) return true;
  if (k.diposting_at) return true;
  if (k.status && k.status !== 'draft') return true;
  return false;
}

async function panggilTeks(prompt) {
  let pesanTerakhir = 'Tidak ada model teks yang bisa dipakai';

  for (let i = 0; i < MODEL_TEKS.length; i++) {
    const nama = MODEL_TEKS[i];
    const alamat =
      'https://generativelanguage.googleapis.com/v1beta/models/' + nama + ':generateContent';

    try {
      const balasan = await fetch(alamat, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'x-goog-api-key': process.env.GEMINI_API_KEY },
        body: JSON.stringify({
          contents: [{ parts: [{ text: prompt }] }],
          generationConfig: {
            responseMimeType: 'application/json',
            maxOutputTokens: 4096,
            temperature: 1,
          },
        }),
      });

      const data = await balasan.json();

      if (!balasan.ok) {
        pesanTerakhir = data && data.error ? data.error.message : 'Gagal (' + nama + ')';
        continue;
      }

      const kandidat = data.candidates && data.candidates[0];
      const bagian = kandidat && kandidat.content ? kandidat.content.parts : [];
      let teks = '';
      for (let j = 0; j < bagian.length; j++) {
        if (bagian[j].text) teks += bagian[j].text;
      }

      if (!teks) {
        pesanTerakhir = 'Model ' + nama + ' tidak mengembalikan teks';
        continue;
      }

      return { hasil: ambilJson(teks), model: nama };
    } catch (e) {
      pesanTerakhir = String(e);
    }
  }

  throw new Error(pesanTerakhir);
}

async function panggilGambar(prompt, fotoBase64, fotoMime, rasio) {
  const alamat =
    'https://generativelanguage.googleapis.com/v1beta/models/' + MODEL_GAMBAR + ':generateContent';

  const bagian = [{ text: prompt }];
  if (fotoBase64) {
    bagian.push({ inline_data: { mime_type: fotoMime || 'image/jpeg', data: fotoBase64 } });
  }

  async function coba(ukuran) {
    const balasan = await fetch(alamat, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-goog-api-key': process.env.GEMINI_API_KEY },
      body: JSON.stringify({
        contents: [{ parts: bagian }],
        generationConfig: {
          responseModalities: ['IMAGE'],
          imageConfig: { imageSize: ukuran, aspectRatio: rasio },
        },
      }),
    });
    const data = await balasan.json();
    return { ok: balasan.ok, data: data };
  }

 let jawab = await coba(UKURAN_GAMBAR);

  // "high demand" biasanya sementara — tunggu sebentar lalu coba sekali lagi
  const sibukSementara = function (j) {
    const p = j.data && j.data.error ? String(j.data.error.message) : '';
    return p.indexOf('high demand') !== -1 ||
           p.indexOf('overloaded') !== -1 ||
           p.indexOf('UNAVAILABLE') !== -1;
  };

  if (!jawab.ok && sibukSementara(jawab)) {
    await new Promise(function (r) { setTimeout(r, 4000); });
    jawab = await coba(UKURAN_GAMBAR);
  }

  if (!jawab.ok && UKURAN_GAMBAR !== '512') jawab = await coba('512');

  if (!jawab.ok) {
    const pesan = jawab.data && jawab.data.error ? jawab.data.error.message : 'Gagal memanggil AI gambar';
    throw new Error(pesan);
  }

  const kandidat = jawab.data.candidates && jawab.data.candidates[0];
  const isi = kandidat && kandidat.content ? kandidat.content.parts : [];

  for (let i = 0; i < isi.length; i++) {
    const inline = isi[i].inlineData || isi[i].inline_data;
    if (inline && inline.data) {
      return { base64: inline.data, mime: inline.mimeType || inline.mime_type || 'image/png' };
    }
  }

  throw new Error('AI tidak mengembalikan gambar');
}

async function unggahGambar(base64, mime) {
  const buffer = Buffer.from(base64, 'base64');
  const ext = mime.indexOf('jpeg') !== -1 ? 'jpg' : 'png';
  const namaFile = 'konten-' + Date.now() + '.' + ext;

  const naik = await admin.storage
    .from('konten')
    .upload(namaFile, buffer, { contentType: mime, upsert: true });

  if (naik.error) throw new Error('Gagal unggah gambar: ' + naik.error.message);

  const pub = admin.storage.from('konten').getPublicUrl(namaFile);
  return pub.data.publicUrl;
}

async function ambilFoto(url) {
  const unduh = await fetch(url);
  if (!unduh.ok) throw new Error('Gagal mengunduh foto produk');
  const ab = await unduh.arrayBuffer();
  return {
    base64: Buffer.from(ab).toString('base64'),
    mime: unduh.headers.get('content-type') || 'image/jpeg',
  };
}

// ambil produk yang paling lama tidak diangkat
async function produkGiliran() {
  const cari = await admin
    .from('produk')
    .select('id, nama, kategori, volume, deskripsi, bahan, tutup, cetak, foto_url, terakhir_diposting')
    .eq('aktif', true)
    .not('foto_url', 'is', null)
    .order('terakhir_diposting', { ascending: true, nullsFirst: true })
    .limit(1);

  if (cari.error) throw new Error('Gagal ambil produk: ' + cari.error.message);
  if (!cari.data || cari.data.length === 0) throw new Error('Tidak ada produk aktif yang punya foto');
  return cari.data[0];
}

          'box makanan, toples — digambarkan lucu dan ramah, boleh diberi wajah sederhana. ' +
          'Suasana ceria dan ringan. Komposisi potret penuh mengisi SELURUH bingkai, ' +
          'tanpa bidang putih polos dan tanpa border. Sepertiga bagian atas dibuat lapang dan ' +
          'sederhana karena akan ditempeli teks. ' +
          'Jangan menambahkan tulisan, huruf, atau angka apa pun di dalam gambar.';
      } else {
        const temaGambar = tipe === 'lepas' && angle
          ? 'Tema: ' + angle + '. '
          : 'Tema: keresahan pelaku UMKM soal tampilan kemasan produknya. ';

        promptGambar =
          'Foto suasana bergaya editorial tentang dunia kemasan produk UMKM Indonesia. ' +
          temaGambar +
          'Contoh adegan: pelaku UMKM sedang mengemas produk, meja kerja dengan bahan kemasan, ' +
          'atau tangan menyerahkan paket ke pembeli. Cahaya alami, warna hangat, terasa nyata. ' +
          'Jangan menampilkan close-up satu produk kemasan tunggal seperti foto katalog. ' +
          'Komposisi potret penuh: foto harus mengisi SELURUH bingkai tanpa area kosong, ' +
          'tanpa bidang putih polos, tanpa border. Sepertiga bagian atas tetap berisi bagian dari pemandangan ' +
          'namun dibuat sederhana dan tidak ramai. ' +
          'Jangan menambahkan tulisan apa pun di dalam gambar.';
      }
    }

    // ---------- JUDUL & PEMBUKA LAMA ----------
    const lama = await admin
      .from('artikel')
      .select('judul')
      .order('dibuat_pada', { ascending: false })
      .limit(25);

    const daftarLama = (lama.data || []).map(function (a) { return a.judul; }).join(' | ') || '(belum ada)';

    const kapLama = await admin
      .from('konten_harian')
      .select('caption, headline_1')
      .order('dibuat_pada', { ascending: false })
      .limit(6);

    const daftarBuka = (kapLama.data || [])
      .map(function (k) { return String(k.caption || '').split('\n')[0].slice(0, 70); })
      .filter(Boolean)
      .join(' / ') || '(belum ada)';

    const daftarH1 = (kapLama.data || [])
      .map(function (k) { return k.headline_1; })
      .filter(Boolean)
      .join(', ') || '(belum ada)';

    const gayaBuka = GAYA_BUKA[Math.floor(Math.random() * GAYA_BUKA.length)];

    // ---------- PROMPT TEKS ----------
    const identitas =
      'Kamu penulis konten untuk "Botol Makassar", toko kemasan di Makassar ' +
      '(botol plastik, botol kaca, gelas, papercup, box makanan, toples, plus jasa sablon & cetak kemasan). ' +
      'Target pembaca: pelaku UMKM makanan & minuman di Indonesia. ' +
      'Gaya: SOFT SELLING — bantu dan edukasi dulu, jualan halus di akhir. ' +
      'Bahasa Indonesia yang santai, hangat, dan jelas. Jangan lebay, jangan pakai kata "revolusioner". ' +
      (momen
        ? 'Konteks waktu sekarang: ' + momen + '. ' +
          'Kaitkan konten dengan momen ini secara wajar dan mengalir — jangan dipaksakan. ' +
          'Kalau tidak nyambung, abaikan saja momen tersebut. '
        : '') +
      'JANGAN pernah mengaitkan konten dengan bencana, kecelakaan, musibah, duka, ' +
      'politik, atau isu sensitif apa pun. Hanya momen positif. ';

    const aturanJson =
      'Balas HANYA dengan JSON valid, tanpa penjelasan dan tanpa tanda kutip markdown. Format persis: ' +
      '{"headline_1":"","headline_2":"","label_ukuran":"","caption":"","judul":"","ringkasan":"","isi":"","angle":"","ringkasan_arc":"","adegan":""} ' +
      'adegan = satu kalimat bahasa Indonesia yang menggambarkan ADEGAN FOTO paling pas untuk isi konten ini ' +
      '(tempat, kegiatan, suasana, waktu). Harus benar-benar mencerminkan judul dan isi kontennya, ' +
      'bukan deskripsi umum. Jangan menyebut tulisan, teks, atau logo apa pun di dalam adegan. ' +
      'Aturan isi: ' +
      'headline_1 = 1 kata pendek yang jadi kata pertama judul di gambar (maksimal 10 huruf). ' +
      'JANGAN memakai kata yang sudah dipakai belakangan ini: ' + daftarH1 + '. ' +
      'headline_2 = lanjutannya, maksimal 25 huruf. ' +
      'label_ukuran = ukuran/volume singkat, atau string kosong kalau tidak relevan. ' +
      'caption = caption Instagram dengan susunan: ' +
      '(1) kalimat pembuka bergaya "' + gayaBuka + '" — WAJIB memakai gaya ini. ' +
      'DILARANG membuka caption dengan frasa klise: "Pernah nggak sih", "Pernah nggak", "Siapa nih", ' +
      '"Buat kamu yang", "Halo Sobat", "Yuk simak", "Tahukah kamu". ' +
      'Kalimat pembuka juga tidak boleh mirip dengan pembuka caption berikut: ' + daftarBuka + '. ' +
      '(2) penjelasan isi konten 3-5 kalimat, (3) satu ajakan halus untuk konsultasi lewat WhatsApp, ' +
      '(4) 5-8 hashtag relevan. Pisahkan tiap bagian dengan baris baru. ' +
      'judul = judul artikel web yang menarik dan berbeda dari daftar judul lama. ' +
      'ringkasan = 1-2 kalimat rangkuman artikel. ' +
      'isi = artikel 400-600 kata dalam format Markdown, pakai beberapa sub-judul "## ", ' +
      'diakhiri ajakan halus untuk konsultasi kemasan lewat WhatsApp Botol Makassar. ' +
      'angle = sudut pandang yang kamu pakai, singkat. ' +
      'ringkasan_arc = 1-2 kalimat padat berisi APA YANG SUDAH DISAMPAIKAN konten ini, ' +
      'ditulis untuk dibaca AI besok sebagai memori penyambung alur. ';

    const dataProduk = produk
      ? 'Data produk:\n' +
        'Nama: ' + (produk.nama || '-') + '\n' +
        'Kategori: ' + (produk.kategori || '-') + '\n' +
        'Volume: ' + (produk.volume || '-') + '\n' +
        'Bahan: ' + (produk.bahan || '-') + '\n' +
        'Tutup: ' + (produk.tutup || '-') + '\n' +
        'Bisa dicetak/sablon: ' + (produk.cetak || '-') + '\n' +
        'Deskripsi: ' + (produk.deskripsi || '-') + '\n' +
        'PENTING: jangan mengarang angka, ukuran, bahan, atau harga di luar data di atas. ' +
        'Jangan menyebutkan harga sama sekali.\n'
      : '';

    let promptTeks;
if (tipe === 'lepas' && jenisLepas === 'interaksi') {
      let khusus;

      if (bentuk === 'quote') {
        khusus =
          'Buat KONTEN QUOTE: satu kalimat pendek yang menyentuh dan membangkitkan semangat ' +
          'pelaku UMKM makanan & minuman — soal ketekunan, melayani pelanggan, atau kebanggaan ' +
          'pada produk sendiri. Harus terasa jujur dan membumi, bukan motivasi kosong. ' +
          'headline_1 & headline_2 = quote-nya dipenggal jadi dua baris. label_ukuran kosong. ' +
          'Susunan caption: (1) quote-nya ditulis utuh, (2) 2-4 kalimat perenungan yang mengaitkannya ' +
          'dengan keseharian usaha kuliner, (3) ajakan berbagi pengalaman di kolom komentar, ' +
          '(4) 5-8 hashtag. JANGAN menjual apa pun di caption ini. ';
      } else if (bentuk === 'tips') {
        khusus =
          'Buat KONTEN TIPS/HACK: satu trik praktis yang bisa langsung dipakai pelaku UMKM ' +
          'makanan & minuman hari itu juga — seputar mengemas, menyimpan, mengirim, atau ' +
          'menampilkan produk. Harus konkret dan spesifik, bukan nasihat umum. ' +
          'headline_1 & headline_2 = inti triknya, singkat. label_ukuran kosong. ' +
          'Susunan caption: (1) satu kalimat masalah yang sering dialami, (2) langkah triknya dalam ' +
          '3-4 poin bernomor pada baris terpisah, (3) ajakan mencoba dan berbagi hasilnya, ' +
          '(4) 5-8 hashtag. ';
      } else {
        khusus =
          'Buat KONTEN KUIS: satu tebak-tebakan ringan seputar kemasan untuk UMKM. ' +
          'Pilihan A, B, dan C WAJIB berupa kategori umum yang bisa dipahami dari teksnya saja — ' +
          'misalnya jenis bahan (PET, HDPE, PP), model tutup, bentuk badan botol, atau ukuran. ' +
          'DILARANG KERAS menyebut nama produk atau nama model tertentu, karena kamu tidak diberi ' +
          'data produk apa pun dan gambarnya tidak menampilkan produk spesifik. ' +
          'headline_1 & headline_2 = PERTANYAAN kuisnya, headline_2 diakhiri tanda tanya. ' +
          'label_ukuran kosong. ' +
          'Susunan caption: (1) satu kalimat pembuka pemancing penasaran, (2) PERTANYAAN kuisnya ' +
          'ditulis utuh dan diakhiri tanda tanya — WAJIB ADA, dilarang langsung melompat ke pilihan, ' +
          '(3) tiga baris pilihan "A. ...", "B. ...", "C. ...", (4) ajakan menjawab di komentar atau DM, ' +
          '(5) 5-8 hashtag. JANGAN membocorkan jawaban benar di caption. ' +
          'isi (artikel) HARUS memuat jawaban benar beserta alasannya. ';
      }

      promptTeks =
        identitas +
        'Konten ini berdiri sendiri, tidak bersambung dengan konten hari lain.\n\n' +
        khusus +
        'PENTING — aturan caption pada bagian "Aturan isi" di bawah TIDAK BERLAKU untuk konten ini; ' +
        'ikuti susunan caption yang baru saja disebutkan. ' +
        'Field "adegan" WAJIB berupa gambaran ilustrasi lucu seputar kemasan yang nyambung dengan ' +
        'isi kontennya, tanpa tulisan apa pun.\n' +
        'Judul artikel lama (jangan mirip): ' + daftarLama + '\n\n' +
        aturanJson;
    } else if (tipe === 'lepas' && jenisLepas === 'kemasan') {

      promptTeks =
        identitas +
        'Tulis konten edukasi tentang KEMASAN secara umum (bukan menyorot satu produk tertentu). ' +
        'Konten ini berdiri sendiri, tidak bersambung dengan konten hari lain. ' +
        'Sudut pandang: ' + angle + '. Pilih topik spesifik yang belum pernah dibahas.\n' +
        'Judul artikel lama (jangan mirip): ' + daftarLama + '\n\n' +
        'headline_1 & headline_2 adalah hook singkat (bukan nama produk), ' +
        'dan label_ukuran diisi string kosong.\n\n' +
        aturanJson;
    } else if (tipe === 'lepas') {
      promptTeks =
        identitas +
        'Tulis konten yang menyorot satu produk ini. ' +
        'Konten ini berdiri sendiri, tidak bersambung dengan konten hari lain — ' +
        'jangan menyinggung "kemarin kita bahas" atau "besok dilanjut".\n\n' +
        dataProduk +
        'headline_1 & headline_2 harus memakai nama produk ini. label_ukuran diambil dari Volume.\n' +
        'Judul artikel lama (jangan mirip): ' + daftarLama + '\n\n' +
        aturanJson;
    } else if (tipe === 'event') {
      promptTeks =
        identitas +
        'Buat konten PERAYAAN untuk momen: ' + (momen || 'perayaan usaha UMKM') + '. ' +
        'Nuansanya hangat dan merayakan, dengan produk di bawah ini sebagai tema visual pendukung. ' +
        'Konten ini berdiri sendiri, bukan bagian dari alur bersambung.\n\n' +
        dataProduk +
        'headline_1 & headline_2 bernuansa perayaan momen tersebut. ' +
        'label_ukuran diambil dari Volume.\n' +
        'Judul artikel lama (jangan mirip): ' + daftarLama + '\n\n' +
        aturanJson;
    } else {
      promptTeks =
        identitas +
        'Konten ini bagian dari satu alur cerita 3 hari yang bersambung. ' +
        instruksiTahap(tahap) + '\n\n' +
        (konteksSebelumnya ? 'MEMORI ALUR SEBELUMNYA:\n' + konteksSebelumnya + '\n' : '') +
        dataProduk +
        (tahap === 'hook'
          ? 'Untuk tahap hook: headline_1 & headline_2 adalah hook masalah (BUKAN nama produk), ' +
            'dan label_ukuran diisi string kosong.\n'
          : 'headline_1 & headline_2 harus memakai nama produk ini. label_ukuran diambil dari Volume.\n') +
        'Judul artikel lama (jangan mirip): ' + daftarLama + '\n\n' +
        aturanJson;
    }

    /// ---------- JALANKAN ----------
    const teks = await panggilTeks(promptTeks);
    const t = teks.hasil;

    // sambungkan gambar dengan isi kontennya
    if (t.adegan) {
      promptGambar +=
       ' Adegan yang harus tergambar (sesuaikan isinya dengan ini, tapi TETAP pertahankan gaya visual ' +
        'yang sudah diminta di atas): ' +
        t.adegan + '.';
    }

    const gambar = await panggilGambar(
      promptGambar, fotoBase64, fotoMime, UKURAN_PILIHAN[namaUkuran].rasio
    );
    const urlGambar = await unggahGambar(gambar.base64, gambar.mime);

    const jenis = pakaiFotoProduk ? 'produk' : 'kemasan';

    // ---------- SIMPAN ----------
 const simpanArtikel = await admin
      .from('artikel')
      .insert({
        judul: t.judul,
        slug: buatSlug(t.judul),
        isi: t.isi,
        ringkasan: t.ringkasan,
        gambar: urlGambar,
        jenis: jenis,
        produk_id: produk ? produk.id : null,
        angle: t.angle,
        tanggal: new Date().toISOString().slice(0, 10),
      })
      .select('id, slug')
      .single();

    if (simpanArtikel.error) throw new Error('Gagal simpan artikel: ' + simpanArtikel.error.message);

    const simpanKonten = await admin
      .from('konten_harian')
      .insert({
        artikel_id: simpanArtikel.data.id,
        produk_id: produk ? produk.id : null,
        jenis: jenis,
        tipe: tipe,
        arc_id: tipe === 'arc' ? arc.id : null,
        tahap: tipe === 'arc' ? tahap : null,
        format: tipe === 'lepas' ? jenisLepas : null,
        rasio: namaUkuran,
        gambar_url: urlGambar,
        headline_1: t.headline_1,
        headline_2: t.headline_2,
        label_ukuran: t.label_ukuran || null,
        caption: t.caption,
        layout: hariKe % 2 === 0 ? 'top' : 'center',
        status: 'draft',
      })
      .select('id')
      .single();

    if (simpanKonten.error) throw new Error('Gagal simpan konten: ' + simpanKonten.error.message);

    // ---------- SIMPAN MEMORI ALUR ----------
    if (tipe === 'arc') {
      const kolom = {
        hook: 'ringkasan_hook',
        build_up: 'ringkasan_build_up',
        closing: 'ringkasan_closing',
      }[tahap];

      const ubah = { tahap: tahap, updated_at: new Date().toISOString() };
      ubah[kolom] = t.ringkasan_arc || t.ringkasan;

      await admin.from('konten_arc').update(ubah).eq('id', arc.id);
    }

    return Response.json({
      sukses: true,
      tipe: tipe,
      tahap: tahap,
      arc_id: tipe === 'arc' ? arc.id : null,
      jenis: jenis,
      rasio: namaUkuran,
      model_teks: teks.model,
      produk: produk ? produk.nama : null,
      judul: t.judul,
      slug: simpanArtikel.data.slug,
      gambar: urlGambar,
      konten_id: simpanKonten.data.id,
    });
  } catch (err) {
    return Response.json({ error: String(err.message || err) }, { status: 500 });
  }
}
